cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(jingyi_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/lib")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_MKL OFF CACHE BOOL "" FORCE)
set(WITH_DNNL OFF CACHE BOOL "" FORCE)
set(WITH_RUY ON CACHE BOOL "" FORCE)
set(WITH_CUDA OFF CACHE BOOL "" FORCE)
set(WITH_CUDNN OFF CACHE BOOL "" FORCE)
set(OPENMP_RUNTIME NONE CACHE STRING "" FORCE)
set(CPUINFO_RUNTIME_TYPE static CACHE STRING "" FORCE)
set(CPUINFO_LIBRARY_TYPE static CACHE STRING "" FORCE)
set(RUY_MINIMAL_BUILD ON CACHE BOOL "" FORCE)
set(CT2_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.cache/native-src/ctranslate2")
add_subdirectory("${CT2_SOURCE}" "${CMAKE_BINARY_DIR}/ctranslate2" EXCLUDE_FROM_ALL)
add_library(jingyi_native STATIC bridge.cpp)
target_link_libraries(jingyi_native PRIVATE ctranslate2)
target_include_directories(jingyi_native PRIVATE "${CT2_SOURCE}/include" "${CT2_SOURCE}/third_party")
target_compile_options(jingyi_native PRIVATE /utf-8 /EHsc)
add_executable(jingyi-native-check EXCLUDE_FROM_ALL check_bridge.cpp)
target_link_libraries(jingyi-native-check PRIVATE jingyi_native)
target_compile_options(jingyi-native-check PRIVATE /utf-8 /EHsc)
